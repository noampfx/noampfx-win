cmake_minimum_required(VERSION 3.22)
project(noamp LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# Build target: MinGW-w64 (GCC) -> noamp.dll, loaded by the JVM through JNI. The same code also builds on Linux (host tests).
option(NOAMP_AVX2 "Use AVX2 + FMA (any PC from ~2013 on). Turn OFF for very old CPUs." ON)

include(FetchContent)
if(POLICY CMP0169)
    cmake_policy(SET CMP0169 OLD)
endif()

# ---- Neural Amp Modeler core (needs internet on the first configure). Pin GIT_TAG to a release for reproducible builds.
FetchContent_Declare(nam_core
    GIT_REPOSITORY https://github.com/sdatkinson/NeuralAmpModelerCore.git
    GIT_TAG main
    GIT_SHALLOW TRUE)
FetchContent_GetProperties(nam_core)
if(NOT nam_core_POPULATED)
    FetchContent_Populate(nam_core)
endif()
file(GLOB_RECURSE NAM_SRC ${nam_core_SOURCE_DIR}/NAM/*.cpp)
set(NAM_INC ${nam_core_SOURCE_DIR})
foreach(d Dependencies/eigen Dependencies/nlohmann Dependencies/nlohmann_json/single_include/nlohmann)
    if(EXISTS ${nam_core_SOURCE_DIR}/${d})
        list(APPEND NAM_INC ${nam_core_SOURCE_DIR}/${d})
    endif()
endforeach()

# ---- miniaudio (single header): WASAPI audio I/O + WAV/MP3/FLAC decoding
FetchContent_Declare(miniaudio
    GIT_REPOSITORY https://github.com/mackron/miniaudio.git
    GIT_TAG 0.11.21
    GIT_SHALLOW TRUE)
FetchContent_GetProperties(miniaudio)
if(NOT miniaudio_POPULATED)
    FetchContent_Populate(miniaudio)
endif()

find_package(JNI REQUIRED)

# ---- optional: ASIO and VST3 (the Steinberg SDKs are not bundled: download them into third_party/, see README_WINDOWS.md)
option(NOAMP_ASIO "ASIO driver support (needs third_party/asiosdk)" ON)
option(NOAMP_VST3 "VST3 plug-in hosting (needs third_party/vst3sdk)" ON)
set(EXTRA_SRC "")
set(EXTRA_INC "")
set(ASIO_DIR ${CMAKE_SOURCE_DIR}/../third_party/asiosdk)
set(VST3_DIR ${CMAKE_SOURCE_DIR}/../third_party/vst3sdk)
# the ASIO SDK keeps its headers either in the root or in common/
foreach(d ${ASIO_DIR} ${ASIO_DIR}/common)
    if(EXISTS ${d}/iasiodrv.h)
        set(ASIO_INC ${d})
    endif()
endforeach()
if(NOAMP_ASIO AND WIN32 AND DEFINED ASIO_INC)
    message(STATUS "ASIO SDK found: ${ASIO_INC}")
    list(APPEND EXTRA_INC ${ASIO_INC})
    set(HAVE_ASIO ON)
else()
    message(STATUS "ASIO disabled (no SDK in third_party/asiosdk)")
endif()
if(NOAMP_VST3 AND WIN32 AND EXISTS ${VST3_DIR}/pluginterfaces/vst/ivstcomponent.h)
    message(STATUS "VST3 SDK found: ${VST3_DIR}")
    list(APPEND EXTRA_INC ${VST3_DIR})
    # interface IIDs are defined in these small files (no need to build the whole SDK)
    foreach(f pluginterfaces/base/coreiids.cpp pluginterfaces/base/funknown.cpp pluginterfaces/base/ustring.cpp pluginterfaces/base/conststringtable.cpp public.sdk/source/vst/vstinitiids.cpp)
        if(EXISTS ${VST3_DIR}/${f})
            list(APPEND EXTRA_SRC ${VST3_DIR}/${f})
        endif()
    endforeach()
    set(HAVE_VST3 ON)
else()
    message(STATUS "VST3 disabled (no SDK in third_party/vst3sdk)")
endif()

add_library(noamp SHARED jni.cpp engine.cpp asio_host.cpp plugin_host.cpp ${EXTRA_SRC} ${NAM_SRC})
set_target_properties(noamp PROPERTIES PREFIX "")
target_include_directories(noamp PRIVATE ${NAM_INC} ${miniaudio_SOURCE_DIR} ${JNI_INCLUDE_DIRS} ${EXTRA_INC})
if(HAVE_ASIO)
    target_compile_definitions(noamp PRIVATE NOAMP_HAS_ASIO)
endif()
if(HAVE_VST3)
    target_compile_definitions(noamp PRIVATE NOAMP_HAS_VST3 RELEASE=1)
endif()
target_compile_definitions(noamp PRIVATE NAM_SAMPLE_FLOAT NDEBUG EIGEN_NO_DEBUG _USE_MATH_DEFINES NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(noamp PRIVATE -O3 -ffast-math -fno-finite-math-only -fno-math-errno -funroll-loops -fomit-frame-pointer -Wno-deprecated-declarations)
if(NOAMP_AVX2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|x86_64|x64")
    target_compile_options(noamp PRIVATE -mavx2 -mfma)
endif()
if(WIN32)
    # fully static: no libstdc++-6.dll / libgcc / winpthread next to the app
    target_link_options(noamp PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_libraries(noamp PRIVATE ole32 winmm uuid advapi32 user32)
else()
    target_link_libraries(noamp PRIVATE pthread dl m)
endif()
